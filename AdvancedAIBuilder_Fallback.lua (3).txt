local InventorySystem = {}

function InventorySystem.UpdateInventory()
    local backpack = player:FindFirstChild("Backpack")
    if not backpack then
        GameState.inventory = {tools = {}, totalItems = 0}
        return
    end
    
    local inventory = {
        tools = {},
        totalItems = 0
    }
    
    for _, item in pairs(backpack:GetChildren()) do
        table.insert(inventory.tools, item.Name)
        inventory.totalItems = inventory.totalItems + 1
    end
    
    GameState.inventory = inventory
end

function InventorySystem.GetInventoryString()
    local inv = GameState.inventory
    local str = "Inventory:\n- Tools: " .. table.concat(inv.tools or {}, ", ") .. "\n- Total: " .. (inv.totalItems or 0)
    return str
end

- Build compact
- Include 3+ different spawn positions

function AISystem.QueryGroq(prompt)
    local systemPrompt = AISystem.GenerateSystemPrompt()
    
    local requestBody = {
        model = "mixtral-8x7b-32768",
        messages = {
            {role = "system", content = systemPrompt},
            {role = "user", content = "Build: " .. prompt}
        },
        max_tokens = 3000,
        temperature = 0.5,
    }
    
    local success, response = pcall(function()
        return HttpService:PostAsync(
            "https://api.groq.com/openai/v1/chat/completions",
            HttpService:JSONEncode(requestBody),
            Enum.HttpContentType.ApplicationJson,
            false,
            {
                ["Content-Type"] = "application/json",
                ["Authorization"] = "Bearer " .. API_KEYS.GROQ
            }
        )
    end)
    
    if not success then return nil end
    
    local decoded = HttpService:JSONDecode(response)
    if decoded.choices and decoded.choices[1] then
        return decoded.choices[1].message.content
    end
    
    return nil
end

function AISystem.QueryHuggingface(prompt)
    local systemPrompt = AISystem.GenerateSystemPrompt()
    
    local requestBody = {
        inputs = systemPrompt .. "\n\nBuild: " .. prompt,
        parameters = {max_length = 3000}
    }
    
    local success, response = pcall(function()
        return HttpService:PostAsync(
            "https://api-inference.huggingface.co/models/mistralai/Mistral-7B-Instruct-v0.2",
            HttpService:JSONEncode(requestBody),
            Enum.HttpContentType.ApplicationJson,
            false,
            {
                ["Content-Type"] = "application/json",
                ["Authorization"] = "Bearer " .. API_KEYS.HUGGINGFACE

        )

    local decoded = HttpService:JSONDecode(response)

    return nil
end

function AISystem.ParseResponse(response)
    if not response then return nil end
    
    local tableStr = response:match("{.-}")
    if not tableStr then return nil end
    
    local success, result = pcall(function()
        return loadstring("return " .. tableStr)()
    end)
    
    if success and result and result.tools then
        return result
    end
    
    return nil
end

function AISystem.QueryAI(prompt)
    print("[AI] Querying Groq...")
    local response = AISystem.QueryGroq(prompt)
    if response then
        local parsed = AISystem.ParseResponse(response)
        if parsed then return parsed, "Groq" end
    end
    
    print("[AI] Trying Huggingface...")
    response = AISystem.QueryHuggingface(prompt)
    if response then
        local parsed = AISystem.ParseResponse(response)
        if parsed then return parsed, "Huggingface" end
    end
    
    print("[AI] Trying Deepseek...")
    response = AISystem.QueryDeepseek(prompt)
    if response then
        local parsed = AISystem.ParseResponse(response)
        if parsed then return parsed, "Deepseek" end
    end
    
    return nil, "none"
end

local BuildExecutor = {}

function BuildExecutor.ExecuteBuild(buildData, centerPosition, statusCallback)
    if not buildData or not buildData.tools then
        return false, "Invalid build data"
    end
    
    GameState.isBuilding = true
    local blocksPlaced = 0
    
    for i, toolData in ipairs(buildData.tools or {}) do
        if not GameState.isBuilding then break end
        if blocksPlaced >= CONFIG.MAX_BLOCKS then break end
        
        local action = toolData.action
        local x = (toolData.x or 0) * CONFIG.BLOCK_SIZE
        local y = (toolData.y or 0) * CONFIG.BLOCK_SIZE
        local z = (toolData.z or 0) * CONFIG.BLOCK_SIZE
        
        local blockPos = centerPosition + Vector3.new(x, y, z)
        
        if action == "spawn_block" then
            ToolSystem.SpawnBlock(blockPos, toolData.material or "brick")
            blocksPlaced = blocksPlaced + 1
            GameState.buildProgress = (blocksPlaced / #(buildData.tools)) * 100
            
            if statusCallback then
                statusCallback(string.format("Building: %d/%d blocks", blocksPlaced, #(buildData.tools)))
            end
        end
        
        wait(0.1)
    end
    
    GameState.isBuilding = false
    return true, "Built " .. blocksPlaced .. " blocks"
end

                StatusLabel:Set("Status: Already building...")

            local prompt = Window.Flags.PromptInput or "house"
            if prompt == "" then
                StatusLabel:Set("Status: ❌ Enter prompt")
                return

            StatusLabel:Set("Status: ✅ Done via " .. service)

    })

    local updateThread = task.spawn(function()
        while true do
            wait(1)
            InventorySystem.UpdateInventory()
            InventoryLabel:Set("Inventory: " .. InventorySystem.GetInventoryString())
            ProgressLabel:Set("Progress: " .. string.format("%.0f%%", GameState.buildProgress))
        end
    end)
